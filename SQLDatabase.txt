-------------------------------------------------------------
-- 1. DROP EXISTING TABLES (In Reverse Dependency Order)
-------------------------------------------------------------
DROP TABLE IF EXISTS MovieImages;
DROP TABLE IF EXISTS MovieGenres;
DROP TABLE IF EXISTS MoviesDb;
DROP TABLE IF EXISTS Certificates;
DROP TABLE IF EXISTS Genres;
DROP TABLE IF EXISTS ScreenSeats;
DROP TABLE IF EXISTS ScreenRows;
DROP TABLE IF EXISTS SeatCategories;
DROP TABLE IF EXISTS ScreenAmenities;
DROP TABLE IF EXISTS Amenities;
DROP TABLE IF EXISTS Screens;
DROP TABLE IF EXISTS TheaterImages;
DROP TABLE IF EXISTS TheaterInformation;
DROP TABLE IF EXISTS Theaters;
DROP TABLE IF EXISTS TheaterAdminProfile;
DROP TABLE IF EXISTS PasswordResetOtps;
DROP TABLE IF EXISTS Media;
DROP TABLE IF EXISTS Users;
DROP TABLE IF EXISTS Roles;

-------------------------------------------------------------
-- 2. CREATE TABLES
-------------------------------------------------------------
-- Roles & Users
CREATE TABLE Roles (
    RoleId UNIQUEIDENTIFIER PRIMARY KEY,
    RoleName NVARCHAR(100) NOT NULL
);

CREATE TABLE Users (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    Username NVARCHAR(255) NOT NULL,
    Email NVARCHAR(255) NOT NULL UNIQUE,
    PasswordHash NVARCHAR(MAX) NOT NULL,
    RoleId UNIQUEIDENTIFIER NOT NULL FOREIGN KEY REFERENCES Roles(RoleId)
);

CREATE TABLE PasswordResetOtps (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    UserId INT NOT NULL FOREIGN KEY REFERENCES Users(Id),
    OtpHash NVARCHAR(MAX) NOT NULL,
    ExpiryTime DATETIME2 NOT NULL,
    IsUsed BIT NOT NULL DEFAULT 0,
    CreatedAt DATETIME2 NOT NULL DEFAULT GETUTCDATE()
);

-- Media
CREATE TABLE Media (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    FilePath NVARCHAR(MAX) NOT NULL,
    FileType NVARCHAR(50),
    FileSize INT,
    UploadedAt DATETIME2 DEFAULT GETUTCDATE()
);

CREATE TABLE TheaterAdminProfile (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    UserId INT NOT NULL FOREIGN KEY REFERENCES Users(Id),
    ProfileImageId INT NULL FOREIGN KEY REFERENCES Media(Id), 
    PhoneNumber NVARCHAR(50),
    Country NVARCHAR(100),
    Status BIT NOT NULL DEFAULT 1,
    CreatedAt DATETIME2 NOT NULL DEFAULT GETUTCDATE()
);

-- Theaters
CREATE TABLE Theaters (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    OwnerUserId INT NOT NULL FOREIGN KEY REFERENCES Users(Id),
    TheaterName NVARCHAR(255) NOT NULL,
    TheaterCode NVARCHAR(50),
    TheaterBrand NVARCHAR(100),
    TheaterDescription NVARCHAR(MAX),
    Status BIT NOT NULL DEFAULT 1
);

CREATE TABLE TheaterInformation (
    TheaterId INT PRIMARY KEY FOREIGN KEY REFERENCES Theaters(Id),
    AddressLine1 NVARCHAR(255),
    AddressLine2 NVARCHAR(255),
    Area NVARCHAR(100),
    City NVARCHAR(100),
    State NVARCHAR(100),
    Country NVARCHAR(100),
    ZipCode NVARCHAR(20),
    ContactName NVARCHAR(255),
    ContactPhone NVARCHAR(50)
);

CREATE TABLE TheaterImages (
    TheaterId INT NOT NULL FOREIGN KEY REFERENCES Theaters(Id),
    MediaId INT NOT NULL FOREIGN KEY REFERENCES Media(Id),
    DisplayOrder INT NOT NULL DEFAULT 0,
    IsPrimary BIT NOT NULL DEFAULT 0,
    PRIMARY KEY (TheaterId, MediaId)
);

-- Screens & Seats
CREATE TABLE Screens (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    TheaterId INT NOT NULL FOREIGN KEY REFERENCES Theaters(Id),
    ScreenName NVARCHAR(255) NOT NULL,
    ScreenCode NVARCHAR(50),
    Floor NVARCHAR(50),
    ScreenType NVARCHAR(50),
    AudioTechnology NVARCHAR(50),
    SeatingType NVARCHAR(50),
    Status BIT NOT NULL DEFAULT 1
);

CREATE TABLE Amenities (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    AmenityName NVARCHAR(100) NOT NULL
);

CREATE TABLE ScreenAmenities (
    ScreenId INT NOT NULL FOREIGN KEY REFERENCES Screens(Id),
    AmenityId INT NOT NULL FOREIGN KEY REFERENCES Amenities(Id),
    PRIMARY KEY (ScreenId, AmenityId)
);

CREATE TABLE SeatCategories (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    CategoryName NVARCHAR(100) NOT NULL,
    ColorCode NVARCHAR(20) NOT NULL
);

CREATE TABLE ScreenRows (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    ScreenId INT NOT NULL FOREIGN KEY REFERENCES Screens(Id),
    RowName NVARCHAR(50) NOT NULL,
    RowType NVARCHAR(50) NOT NULL,
    TotalSeats INT NOT NULL,
    WalkwayCount INT NOT NULL DEFAULT 0,
    SeatCategoryId INT FOREIGN KEY REFERENCES SeatCategories(Id),
    DisplayOrder INT NOT NULL DEFAULT 0,
    IsReverse BIT NOT NULL DEFAULT 0,
    Status BIT NOT NULL DEFAULT 1
);

CREATE TABLE ScreenSeats (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    ScreenRowId INT NOT NULL FOREIGN KEY REFERENCES ScreenRows(Id),
    SeatNo INT NOT NULL,
    SeatLabel NVARCHAR(50) NOT NULL,
    SeatType NVARCHAR(50) NOT NULL,
    Status BIT NOT NULL DEFAULT 1
);

-- Movies
CREATE TABLE Certificates (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    CertificateName NVARCHAR(100) NOT NULL
);

CREATE TABLE Genres (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    GenreName NVARCHAR(100) NOT NULL
);

CREATE TABLE MoviesDb (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    TheaterId INT NOT NULL FOREIGN KEY REFERENCES Theaters(Id),
    MovieName NVARCHAR(255) NOT NULL,
    MovieCode NVARCHAR(50),
    MovieDescription NVARCHAR(MAX),
    TrailerUrl NVARCHAR(MAX),
    TeaserUrl NVARCHAR(MAX),
    Duration INT NOT NULL, -- Duration in minutes
    ReleaseDate DATE,
    CertificateId INT FOREIGN KEY REFERENCES Certificates(Id),
    Status BIT NOT NULL DEFAULT 1,
    Tags NVARCHAR(MAX)
);

CREATE TABLE MovieGenres (
    MovieId INT NOT NULL FOREIGN KEY REFERENCES MoviesDb(Id),
    GenreId INT NOT NULL FOREIGN KEY REFERENCES Genres(Id),
    PRIMARY KEY (MovieId, GenreId)
);

CREATE TABLE MovieImages (
    MovieId INT NOT NULL FOREIGN KEY REFERENCES MoviesDb(Id),
    MediaId INT NOT NULL FOREIGN KEY REFERENCES Media(Id),
    DisplayOrder INT NOT NULL DEFAULT 0,
    IsPrimary BIT NOT NULL DEFAULT 0,
    PRIMARY KEY (MovieId, MediaId)
);
